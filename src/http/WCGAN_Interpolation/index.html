C++ Network & DAG Code

Generator (Residual + Bilinear Upsampling + BatchNorm) & Discriminator

Allocator a;

auto res_block = [](Node* X, size_t channels) {
  Node* h = a.Deconvolution2D(X, {3, 3}, channels, 1);
  h = a.BatchNormalization(h);
  h = a.LeakyRelu(h);
  h = a.Convolution2D(h, {3, 3}, channels, 1);
  h = a.BatchNormalization(h);
  return a.LeakyRelu(a.Add(X, h));
};

auto generator = [](Node* X) {
  X = a.Linear(X, {4, 4, 32});
  X = a.BatchNormalization(X);
  X = a.LeakyRelu(X);

  X = a.BilinearUpsampling(X);            // 9x9x32
  X = a.Convolution2D(X, {2, 2}, 24, 1);  // 8x8x24
  X = a.BatchNormalization(X);
  X = a.LeakyRelu(X);

  X = res_block(X, 24);                   // 8x8x24 residual

  X = a.BilinearUpsampling(X);            // 17x17x24
  X = a.Convolution2D(X, {3, 3}, 16, 1);  // 15x15x16
  X = a.BatchNormalization(X);
  X = a.LeakyRelu(X);

  X = a.BilinearUpsampling(X);            // 31x31x16
  X = a.Convolution2D(X, {3, 3}, 1, 1);   // 29x29x1
  X = a.Sigmoid(X);
  return X;
};

auto discriminator = [](Node* X) {
  X = a.Convolution2D(X, {4, 4}, 16, 1);  // 26x26x16
  X = a.LeakyRelu(X);

  X = a.Convolution2D(X, {4, 4}, 24, 2);  // 12x12x24
  X = a.LeakyRelu(X);

  X = a.Convolution2D(X, {4, 4}, 32, 2);  // 5x5x32
  X = a.LeakyRelu(X);

  X = a.Linear(X, {1, 1, 1});             // real vs fake
  X = a.Sigmoid(X);
  return X;
};

auto gen_in  = a.Input({10});             auto gen_out  = generator(gen_in);
auto disc_in = a.Input({29, 29, 1});      auto disc_out = discriminator(disc_in);
Node* disc_first = disc_in->outputs[0];

Adversarial Step with Independent Optimizers & DAG Rewiring

Graph gen_graph(gen_out), disc_graph(disc_out);
Adam opt_g(gen_graph.TrainableNodes(), 0.5f);  // DCGAN beta1 = 0.5
Adam opt_d(disc_graph.TrainableNodes(), 0.5f);

// Generator step: plug generator output directly into discriminator,
// backpropagate through both, and step ONLY opt_g.
disc_first->SetInput(0, gen_out);
Graph full_graph(disc_out);
full_graph.Forward(batch, true);
full_graph.ZeroGrad(batch);

full_graph.Backward(batch);
